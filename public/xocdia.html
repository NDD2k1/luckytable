bindSqueezeGestures() { const bowl = document.getElementById('bowl-cover'); const dishStage = document.getElementById('dish-stage'); const onStart = (clientX, clientY) => { if (this.state !== GameState.SQUEEZING) return; this.isDragging = true; this.dragStartX = clientX - this.cupOffset.x; this.dragStartY = clientY - this.cupOffset.y; bowl.classList.add('squeezing-active'); }; const onMove = (clientX, clientY) => { if (!this.isDragging || this.state !== GameState.SQUEEZING) return; const x = clientX - this.dragStartX; const y = clientY - this.dragStartY; this.cupOffset = { x, y }; // Di chuyển bát sứ theo tay kéo tự do 360 độ bowl.style.transform = `translate(${x}px, ${y}px) scale(1.03)`; const dist = Math.sqrt(x * x + y * y); // Ngưỡng kéo mở hoàn toàn: 130px if (dist > 130) { this.isDragging = false; bowl.classList.remove('squeezing-active'); this.endSqueezeAndReveal(); } }; const onEnd = () => { if (!this.isDragging || this.state !== GameState.SQUEEZING) return; this.isDragging = false; bowl.classList.remove('squeezing-active'); const dist = Math.sqrt(this.cupOffset.x * this.cupOffset.x + this.cupOffset.y * this.cupOffset.y); if (dist > 115) { this.endSqueezeAndReveal(); } }; // Chuột máy tính (Desktop) bowl.addEventListener('mousedown', (e) => { e.preventDefault(); onStart(e.clientX, e.clientY); }); window.addEventListener('mousemove', (e) => onMove(e.clientX, e.clientY)); window.addEventListener('mouseup', onEnd); // Màn hình cảm ứng (Mobile / Tablet) bowl.addEventListener('touchstart', (e) => { if (e.touches && e.touches[0]) { e.preventDefault(); onStart(e.touches[0].clientX, e.touches[0].clientY); } }, { passive: false }); window.addEventListener('touchmove', (e) => { if (e.touches && e.touches[0]) { onMove(e.touches[0].clientX, e.touches[0].clientY); } }, { passive: true }); window.addEventListener('touchend', onEnd); }
